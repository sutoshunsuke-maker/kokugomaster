// ログイン
function login() {
  const userId = document.getElementById('user-id').value;
  const userPass = document.getElementById('user-pass').value;

  if (userId === 'teacher' && userPass === '1234') {
    currentUser = { id: 'teacher', role: 'teacher' };
    showScreen('teacher-screen');
    renderTeacherDashboard();
    return;
  }

  const student = students.find(s => s.id === userId && s.pass === userPass);
  if (student) {
    currentUser = student;
    
    // アバター未選択の場合は選択モーダルを表示
    if (!student.avatar) {
      document.getElementById('avatar-select-modal').classList.remove('hidden');
      return;
    }

    startStudentSession();
    return;
  }
  alert('IDまたはパスワードが違います');
}

// 初回アバター選択
function selectAvatar(avatarPath) {
  currentUser.avatar = avatarPath;
  document.getElementById('avatar-select-modal').classList.add('hidden');
  startStudentSession();
}

function startStudentSession() {
  let gotBonus = false;
  if (isClassMode && !currentUser.checkedInToday) {
    currentUser.currentPt += 1;
    currentUser.totalPt += 1;
    currentUser.todayPt += 1;
    currentUser.checkedInToday = true;
    gotBonus = true;
  }
  showScreen('student-screen');
  renderStudentDashboard(gotBonus);
}

function logout() {
  currentUser = null;
  document.getElementById('user-id').value = '';
  document.getElementById('user-pass').value = '';
  showScreen('login-screen');
}

function showScreen(screenId) {
  ['login-screen', 'teacher-screen', 'student-screen', 'adventure-screen'].forEach(id => {
    document.getElementById(id).classList.add('hidden');
  });
  document.getElementById(screenId).classList.remove('hidden');
}

// 教員用ダッシュボード（アバター反映 ＆ 柔軟なPt付与）
function renderTeacherDashboard() {
  const grid = document.getElementById('student-grid');
  grid.innerHTML = '';
  students.forEach(st => {
    const card = document.createElement('div');
    card.className = `student-card ${st.todayPt === 0 ? 'low-pt' : ''}`;
    
    const avatarHTML = st.avatar 
      ? `<div class="img-crop"><img src="${st.avatar}"></div>`
      : `<div class="img-crop" style="display:flex;align-items:center;justify-content:center;color:#aaa;">未設定</div>`;

    card.innerHTML = `
      ${avatarHTML}
      <h4>${st.name}</h4>
      <p>本日獲得: <b>+${st.todayPt} pt</b></p>
      <p><small>所持: ${st.currentPt}pt / 累計: ${st.totalPt}pt</small></p>
      <div class="pt-buttons">
        <button onclick="addPoint('${st.id}', 1)">+1pt</button>
        <button onclick="addPoint('${st.id}', 5)">+5pt</button>
        <button onclick="addPoint('${st.id}', 10)">+10pt</button>
      </div>
      <div class="custom-pt-box">
        <input type="number" id="custom-pt-${st.id}" placeholder="任意pt">
        <button onclick="addCustomPoint('${st.id}')">加算</button>
      </div>
    `;
    grid.appendChild(card);
  });
}

// ポイント加算（確定処理）
function addPoint(studentId, pt) {
  const st = students.find(s => s.id === studentId);
  if (st) {
    st.currentPt += pt;
    st.totalPt += pt;
    st.todayPt += pt;
    renderTeacherDashboard();
  }
}

// 任意数値のポイント加算
function addCustomPoint(studentId) {
  const input = document.getElementById(`custom-pt-${studentId}`);
  const val = parseInt(input.value);
  if (!isNaN(val) && val > 0) {
    addPoint(studentId, val);
    input.value = '';
  }
}

// 新しい単元・セリフ・スキルの追加機能
function addNewStage() {
  const title = document.getElementById('new-stage-title').value.trim();
  const dialogueText = document.getElementById('new-stage-dialogue').value.trim();
  const skill = document.getElementById('new-stage-skill').value.trim();

  if (!title || !dialogueText || !skill) {
    alert('すべての項目を入力してください');
    return;
  }

  // 改行テキストを「話者」と「本文」のオブジェクト配列に変換
  const lines = dialogueText.split('\n');
  const parsedDialogues = lines.map(line => {
    const parts = line.split(':');
    if (parts.length >= 2) {
      return { speaker: parts[0].trim(), text: parts.slice(1).join(':').trim() };
    } else {
      return { speaker: 'システム', text: line.trim() };
    }
  });

  stages.push({
    title: title,
    skill: skill,
    dialogues: parsedDialogues
  });

  alert(`新しい単元「${title}」を追加しました！`);
  document.getElementById('new-stage-title').value = '';
  document.getElementById('new-stage-dialogue').value = '';
  document.getElementById('new-stage-skill').value = '';
}

function toggleClassMode() {
  isClassMode = !isClassMode;
  document.getElementById('mode-text').innerText = isClassMode ? '授業中 (ON)' : '授業外 (OFF)';
  document.getElementById('mode-text').className = isClassMode ? 'text-on' : 'text-off';
  document.getElementById('mode-btn').innerText = isClassMode ? '授業を終了する (OFFにする)' : '授業を開始する (ONにする)';
}

function sortStudentsByToday() {
  students.sort((a, b) => a.todayPt - b.todayPt);
  renderTeacherDashboard();
}

// 生徒画面更新
function renderStudentDashboard(gotBonus) {
  document.getElementById('st-name').innerText = currentUser.name;
  document.getElementById('st-pt').innerText = currentUser.currentPt;
  document.getElementById('st-total').innerText = currentUser.totalPt;
  document.getElementById('equipped-item').innerText = `装備: ${currentUser.equipped}`;

  // 生徒アバター画像表示
  const avatarContainer = document.getElementById('st-avatar-img');
  avatarContainer.innerHTML = `<img src="${currentUser.avatar}">`;

  const bonusBanner = document.getElementById('bonus-banner');
  if (gotBonus) bonusBanner.classList.remove('hidden');
  else bonusBanner.classList.add('hidden');

  // 授業中の着脱制限メッセージ
  const lockMsg = document.getElementById('class-lock-msg');
  if (isClassMode) lockMsg.classList.remove('hidden');
  else lockMsg.classList.add('hidden');

  // もちもの表示
  const itemList = document.getElementById('item-list');
  itemList.innerHTML = '';
  if (currentUser.items.length === 0) {
    itemList.innerHTML = '<li class="empty">持っているアイテムはありません</li>';
  } else {
    currentUser.items.forEach(item => {
      const li = document.createElement('li');
      const isEquipped = currentUser.equipped === item;
      li.innerHTML = `
        <span>🎁 ${item} ${isEquipped ? '<b>(装備中)</b>' : ''}</span>
        ${!isClassMode ? `<button onclick="toggleEquip('${item}')">${isEquipped ? '外す' : '装備する'}</button>` : ''}
      `;
      itemList.appendChild(li);
    });
  }

  // スキル表示
  const skillList = document.getElementById('skill-list');
  skillList.innerHTML = '';
  if (currentUser.skills.length === 0) {
    skillList.innerHTML = '<li class="empty">まだ習得している技はありません</li>';
  } else {
    currentUser.skills.forEach(sk => {
      const li = document.createElement('li');
      li.innerText = `🗡️ 【技】${sk}`;
      skillList.appendChild(li);
    });
  }

  // マップ（教員が追加した単元も一覧で動的生成）
  const mapContainer = document.getElementById('map-container');
  mapContainer.innerHTML = '';
  stages.forEach(stg => {
    const div = document.createElement('div');
    div.className = 'stage-building';
    div.onclick = () => startAdventure(stg.title);
    div.innerHTML = `🏛️<br><b>${stg.title}</b>`;
    mapContainer.appendChild(div);
  });
}

function toggleEquip(itemName) {
  if (isClassMode) return;
  currentUser.equipped = (currentUser.equipped === itemName) ? 'なし' : itemName;
  renderStudentDashboard(false);
}

// アドベンチャー進行
function startAdventure(stageTitle) {
  currentStage = stages.find(s => s.title === stageTitle);
  currentDialogueIndex = 0;
  document.getElementById('adv-title').innerText = `単元: ${currentStage.title}`;
  showScreen('adventure-screen');
  updateDialogue();
}

function updateDialogue() {
  const current = currentStage.dialogues[currentDialogueIndex];
  document.getElementById('speaker-name').innerText = current.speaker;
  document.getElementById('dialogue-text').innerText = current.text;
  document.getElementById('next-dialogue-btn').innerText = (currentDialogueIndex === currentStage.dialogues.length - 1) ? 'ステージクリア！' : '次へ ▶';
}

function nextDialogue() {
  if (currentDialogueIndex < currentStage.dialogues.length - 1) {
    currentDialogueIndex++;
    updateDialogue();
  } else {
    // クリア時のスキル獲得
    if (!currentUser.skills.includes(currentStage.skill)) {
      currentUser.skills.push(currentStage.skill);
      alert(`✨ クリア！ スキル【${currentStage.skill}】を新しく習得しました！`);
    } else {
      alert(`🎉 ステージクリア！`);
    }
    showScreen('student-screen');
    renderStudentDashboard(false);
  }
}

// ガチャ
function openGachaModal() {
  const lockMsg = document.getElementById('gacha-lock-msg');
  const body = document.getElementById('gacha-body');

  if (isClassMode) {
    lockMsg.classList.remove('hidden');
    body.classList.add('hidden');
  } else {
    lockMsg.classList.add('hidden');
    body.classList.remove('hidden');
    document.getElementById('gacha-user-pt').innerText = currentUser.currentPt;
    document.getElementById('gacha-result').classList.add('hidden');
  }
  document.getElementById('gacha-modal').classList.remove('hidden');
}

function closeGachaModal() {
  document.getElementById('gacha-modal').classList.hidden = true;
  document.getElementById('gacha-modal').classList.add('hidden');
  renderStudentDashboard(false);
}

function drawRarityItem(pool) {
  const totalWeight = pool.reduce((sum, i) => sum + i.weight, 0);
  let random = Math.random() * totalWeight;
  for (const item of pool) {
    if (random < item.weight) return item.name;
    random -= item.weight;
  }
  return pool[0].name;
}

function playGacha(type) {
  const cost = type === 'normal' ? 10 : 20;
  if (currentUser.currentPt < cost) {
    alert('ポイントが足りません！');
    return;
  }

  currentUser.currentPt -= cost;
  document.getElementById('gacha-user-pt').innerText = currentUser.currentPt;

  const pool = type === 'normal' ? normalGachaPool : limitedGachaPool;
  const reward = drawRarityItem(pool);

  currentUser.items.push(reward);

  const resultBox = document.getElementById('gacha-result');
  resultBox.innerText = `🎉 ガチャ結果: 「${reward}」 を獲得しました！（もちものに追加）`;
  resultBox.classList.remove('hidden');
}
