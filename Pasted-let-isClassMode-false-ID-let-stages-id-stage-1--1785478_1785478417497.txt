// ログイン
function login() {
  const userId = document.getElementById('user-id').value;
  const userPass = document.getElementById('user-pass').value;

  if (userId === 'teacher' && userPass === '1234') {
    currentUser = { id: 'teacher', role: 'teacher' };
    showScreen('teacher-screen');
    renderTeacherDashboard();
    return;
  }

  const student = students.find(s => s.id === userId && s.pass === userPass);
  if (student) {
    currentUser = student;
    if (!student.avatar) {
      document.getElementById('avatar-select-modal').classList.remove('hidden');
      return;
    }
    startStudentSession();
    return;
  }
  alert('IDまたはパスワードが違います');
}

function selectAvatar(avatarPath) {
  currentUser.avatar = avatarPath;
  document.getElementById('avatar-select-modal').classList.add('hidden');
  startStudentSession();
}

function startStudentSession() {
  let gotBonus = false;
  if (isClassMode && !currentUser.checkedInToday) {
    currentUser.currentPt += 1;
    currentUser.totalPt += 1;
    currentUser.todayPt += 1;
    currentUser.checkedInToday = true;
    gotBonus = true;
  }
  showScreen('student-screen');
  renderStudentDashboard(gotBonus);
}

function logout() {
  currentUser = null;
  document.getElementById('user-id').value = '';
  document.getElementById('user-pass').value = '';
  showScreen('login-screen');
}

function showScreen(screenId) {
  ['login-screen', 'teacher-screen', 'student-screen', 'adventure-screen'].forEach(id => {
    document.getElementById(id).classList.add('hidden');
  });
  document.getElementById(screenId).classList.remove('hidden');
}

// 教員画面：生徒一覧とミッションクリア操作
function renderStudentGrid() {
  const grid = document.getElementById('student-grid');
  grid.innerHTML = '';
  
  students.forEach(st => {
    const card = document.createElement('div');
    card.className = `student-card ${st.todayPt === 0 ? 'low-pt' : ''}`;
    
    const avatarHTML = st.avatar 
      ? `<div class="img-crop"><img src="${st.avatar}"></div>`
      : `<div class="img-crop" style="display:flex;align-items:center;justify-content:center;color:#aaa;">未設定</div>`;

    // 生徒ごとの各単元ミッションクリア状況のドロップダウン・ボタン
    let missionControlsHTML = '';
    stages.forEach(stg => {
      const currentProg = st.progress[stg.id] || 0;
      const totalMissions = stg.missions.length;
      const isCompleted = currentProg >= totalMissions && totalMissions > 0;

      missionControlsHTML += `
        <div class="student-mission-box">
          <b>${stg.title}</b>: 
          ${isCompleted ? '<span style="color:#48bb78;">全クリア!</span>' : `${currentProg}/${totalMissions} 完了`}
          <div style="margin-top: 5px; display:flex; gap:5px; align-items:center;">
            <button onclick="advanceStudentMission('${st.id}', '${stg.id}')" ${isCompleted ? 'disabled' : ''}>
              ${isCompleted ? 'クリア済み' : '次のミッション達成!'}
            </button>
            <button onclick="resetStudentMission('${st.id}', '${stg.id}')" class="btn-secondary" style="font-size:0.7em;">リセット</button>
          </div>
        </div>
      `;
    });

    card.innerHTML = `
      ${avatarHTML}
      <h4>${st.name}</h4>
      <p>本日獲得: <b>+${st.todayPt} pt</b></p>
      <p><small>所持: ${st.currentPt}pt / 累計: ${st.totalPt}pt</small></p>
      
      <div class="pt-buttons">
        <button onclick="addPoint('${st.id}', 1)">+1pt</button>
        <button onclick="addPoint('${st.id}', 5)">+5pt</button>
        <button onclick="addPoint('${st.id}', 10)">+10pt</button>
      </div>
      <div class="custom-pt-box">
        <input type="number" id="custom-pt-${st.id}" placeholder="pt">
        <button onclick="addCustomPoint('${st.id}')">加算</button>
      </div>

      <hr style="border:0; border-top:1px solid #5a5a7a; margin:10px 0;">
      <h5>🎯 ミッション進捗の更新</h5>
      ${missionControlsHTML}
    `;
    grid.appendChild(card);
  });
}

// 教員操作：生徒のミッションを1つ進める
function advanceStudentMission(studentId, stageId) {
  const st = students.find(s => s.id === studentId);
  const stg = stages.find(s => s.id === stageId);
  if (!st || !stg) return;

  if (!st.progress[stageId]) st.progress[stageId] = 0;
  
  if (st.progress[stageId] < stg.missions.length) {
    st.progress[stageId]++;
    
    // 全ミッション完了時にスキル自動習得処理
    if (st.progress[stageId] === stg.missions.length) {
      if (!st.skills.includes(stg.skill)) {
        st.skills.push(stg.skill);
      }
    }
  }
  renderStudentGrid();
}

// 教員操作：ミッション進捗のリセット
function resetStudentMission(studentId, stageId) {
  const st = students.find(s => s.id === studentId);
  if (st) {
    st.progress[stageId] = 0;
    renderStudentGrid();
  }
}

// ポイント操作
function addPoint(studentId, pt) {
  const st = students.find(s => s.id === studentId);
  if (st) {
    st.currentPt += pt;
    st.totalPt += pt;
    st.todayPt += pt;
    renderStudentGrid();
  }
}

function addCustomPoint(studentId) {
  const input = document.getElementById(`custom-pt-${studentId}`);
  const val = parseInt(input.value);
  if (!isNaN(val) && val > 0) {
    addPoint(studentId, val);
    input.value = '';
  }
}

// 単元（学習マップ）の保存・更新
function saveStage() {
  const editId = document.getElementById('editing-stage-id').value;
  const title = document.getElementById('stage-title').value.trim();
  const missionsText = document.getElementById('stage-missions').value.trim();
  const dialogueText = document.getElementById('stage-dialogue').value.trim();
  const skill = document.getElementById('stage-skill').value.trim();

  if (!title || !dialogueText || !skill) {
    alert('タイトル・会話・スキルを入力してください');
    return;
  }

  // ミッション（改行区切り配列化）
  const missions = missionsText ? missionsText.split('\n').map(m => m.trim()).filter(m => m.length > 0) : ['① 学習を完了する'];

  // 会話テキスト解析
  const parsedDialogues = dialogueText.split('\n').map(line => {
    const parts = line.split(':');
    if (parts.length >= 2) {
      return { speaker: parts[0].trim(), text: parts.slice(1).join(':').trim() };
    } else {
      return { speaker: '語り', text: line.trim() };
    }
  });

  if (editId) {
    // 編集更新
    const stg = stages.find(s => s.id === editId);
    if (stg) {
      stg.title = title;
      stg.missions = missions;
      stg.dialogues = parsedDialogues;
      stg.skill = skill;
      alert(`単元「${title}」を更新しました！`);
    }
  } else {
    // 新規登録
    const newId = 'stage_' + Date.now();
    stages.push({
      id: newId,
      title: title,
      missions: missions,
      dialogues: parsedDialogues,
      skill: skill
    });
    alert(`新しい単元「${title}」を登録しました！`);
  }

  resetStageForm();
  renderTeacherDashboard();
}

// 編集フォームにデータを読み込む
function editStage(stageId) {
  const stg = stages.find(s => s.id === stageId);
  if (!stg) return;

  document.getElementById('stage-form-title').innerText = '✏️ 単元・ミッションの編集';
  document.getElementById('editing-stage-id').value = stg.id;
  document.getElementById('stage-title').value = stg.title;
  document.getElementById('stage-missions').value = stg.missions.join('\n');
  document.getElementById('stage-dialogue').value = stg.dialogues.map(d => `${d.speaker}: ${d.text}`).join('\n');
  document.getElementById('stage-skill').value = stg.skill;

  document.getElementById('save-stage-btn').innerText = '更新内容を保存する';
  document.getElementById('cancel-edit-btn').classList.remove('hidden');
}

// フォームのリセット
function resetStageForm() {
  document.getElementById('stage-form-title').innerText = '📖 単元・ミッションの設定';
  document.getElementById('editing-stage-id').value = '';
  document.getElementById('stage-title').value = '';
  document.getElementById('stage-missions').value = '';
  document.getElementById('stage-dialogue').value = '';
  document.getElementById('stage-skill').value = '';

  document.getElementById('save-stage-btn').innerText = '単元を登録・更新する';
  document.getElementById('cancel-edit-btn').classList.add('hidden');
}

// 単元の削除
function deleteStage(stageId) {
  if (confirm('本当にこの単元を削除しますか？')) {
    stages = stages.filter(s => s.id !== stageId);
    renderTeacherDashboard();
  }
}

// 配布中単元リストの描画（教員画面）
function renderTeacherStageList() {
  const list = document.getElementById('teacher-stage-list');
  list.innerHTML = '';

  if (stages.length === 0) {
    list.innerHTML = '<p class="hint">現在配布されている単元はありません。</p>';
    return;
  }

  stages.forEach(stg => {
    const div = document.createElement('div');
    div.className = 'teacher-stage-item';
    div.innerHTML = `
      <div>
        <strong>📖 ${stg.title}</strong> (獲得スキル: ${stg.skill})
        <br><small style="color:#a0aec0;">ミッション数: ${stg.missions.length} / 会話数: ${stg.dialogues.length}</small>
      </div>
      <div>
        <button onclick="editStage('${stg.id}')">編集</button>
        <button onclick="deleteStage('${stg.id}')" class="btn-danger">削除</button>
      </div>
    `;
    list.appendChild(div);
  });
}

// 生徒画面更新
function renderStudentDashboard(gotBonus) {
  document.getElementById('st-name').innerText = currentUser.name;
  document.getElementById('st-pt').innerText = currentUser.currentPt;
  document.getElementById('st-total').innerText = currentUser.totalPt;
  document.getElementById('equipped-item').innerText = `装備: ${currentUser.equipped}`;

  const avatarContainer = document.getElementById('st-avatar-img');
  avatarContainer.innerHTML = `<img src="${currentUser.avatar}">`;

  const bonusBanner = document.getElementById('bonus-banner');
  if (gotBonus) bonusBanner.classList.remove('hidden');
  else bonusBanner.classList.add('hidden');

  const lockMsg = document.getElementById('class-lock-msg');
  if (isClassMode) lockMsg.classList.remove('hidden');
  else lockMsg.classList.add('hidden');

  // もちもの
  const itemList = document.getElementById('item-list');
  itemList.innerHTML = '';
  if (currentUser.items.length === 0) {
    itemList.innerHTML = '<li class="empty">持っているアイテムはありません</li>';
  } else {
    currentUser.items.forEach(item => {
      const li = document.createElement('li');
      const isEquipped = currentUser.equipped === item;
      li.innerHTML = `
        <span>🎁 ${item} ${isEquipped ? '<b>(装備中)</b>' : ''}</span>
        ${!isClassMode ? `<button onclick="toggleEquip('${item}')">${isEquipped ? '外す' : '装備する'}</button>` : ''}
      `;
      itemList.appendChild(li);
    });
  }

  // スキル
  const skillList = document.getElementById('skill-list');
  skillList.innerHTML = '';
  if (currentUser.skills.length === 0) {
    skillList.innerHTML = '<li class="empty">まだ習得している技はありません</li>';
  } else {
    currentUser.skills.forEach(sk => {
      const li = document.createElement('li');
      li.innerText = `🗡️ 【技】${sk}`;
      skillList.appendChild(li);
    });
  }

  // マップ（動的生成）
  const mapContainer = document.getElementById('map-container');
  mapContainer.innerHTML = '';
  stages.forEach(stg => {
    const div = document.createElement('div');
    div.className = 'stage-building';
    div.onclick = () => startAdventure(stg.id);
    
    const prog = currentUser.progress[stg.id] || 0;
    const isCompleted = prog >= stg.missions.length;

    div.innerHTML = `
      🏛️<br><b>${stg.title}</b><br>
      <small style="color:${isCompleted ? '#48bb78' : '#ecc94b'};">
        ${isCompleted ? 'クリア済み' : `進行中 (${prog}/${stg.missions.length})`}
      </small>
    `;
    mapContainer.appendChild(div);
  });
}

function toggleEquip(itemName) {
  if (isClassMode) return;
  currentUser.equipped = (currentUser.equipped === itemName) ? 'なし' : itemName;
  renderStudentDashboard(false);
}

// アドベンチャー（物語体験）の開始
function startAdventure(stageId) {
  currentStage = stages.find(s => s.id === stageId);
  currentDialogueIndex = 0;
  document.getElementById('adv-title').innerText = `単元: ${currentStage.title}`;
  
  showScreen('adventure-screen');
  renderMissionStatus();
  updateDialogue();
}

// 生徒アドベンチャー画面のミッション状況描画
function renderMissionStatus() {
  const container = document.getElementById('adv-mission-list');
  container.innerHTML = '';

  const userProg = currentUser.progress[currentStage.id] || 0;

  currentStage.missions.forEach((mText, idx) => {
    const isCleared = idx < userProg;
    const div = document.createElement('div');
    div.className = `mission-item ${isCleared ? 'cleared' : 'locked'}`;
    div.innerText = `${isCleared ? '✅' : '🔒'} ${mText}`;
    container.appendChild(div);
  });
}

function updateDialogue() {
  const current = currentStage.dialogues[currentDialogueIndex];
  document.getElementById('speaker-name').innerText = current.speaker;
  document.getElementById('dialogue-text').innerText = current.text;
  document.getElementById('next-dialogue-btn').innerText = (currentDialogueIndex === currentStage.dialogues.length - 1) ? 'もう一度読む' : '次へ ▶';
}

function nextDialogue() {
  if (currentDialogueIndex < currentStage.dialogues.length - 1) {
    currentDialogueIndex++;
    updateDialogue();
  } else {
    currentDialogueIndex = 0;
    updateDialogue();
  }
}

// ガチャ機能
function openGachaModal() {
  const lockMsg = document.getElementById('gacha-lock-msg');
  const body = document.getElementById('gacha-body');

  if (isClassMode) {
    lockMsg.classList.remove('hidden');
    body.classList.add('hidden');
  } else {
    lockMsg.classList.add('hidden');
    body.classList.remove('hidden');
    document.getElementById('gacha-user-pt').innerText = currentUser.currentPt;
    document.getElementById('gacha-result').classList.add('hidden');
  }
  document.getElementById('gacha-modal').classList.remove('hidden');
}

function closeGachaModal() {
  document.getElementById('gacha-modal').classList.add('hidden');
  renderStudentDashboard(false);
}

function drawRarityItem(pool) {
  const totalWeight = pool.reduce((sum, i) => sum + i.weight, 0);
  let random = Math.random() * totalWeight;
  for (const item of pool) {
    if (random < item.weight) return item.name;
    random -= item.weight;
  }
  return pool[0].name;
}

function playGacha(type) {
  const cost = type === 'normal' ? 10 : 20;
  if (currentUser.currentPt < cost) {
    alert('ポイントが足りません！');
    return;
  }

  currentUser.currentPt -= cost;
  document.getElementById('gacha-user-pt').innerText = currentUser.currentPt;

  const pool = type === 'normal' ? normalGachaPool : limitedGachaPool;
  const reward = drawRarityItem(pool);

  currentUser.items.push(reward);

  const resultBox = document.getElementById('gacha-result');
  resultBox.innerText = `🎉 ガチャ結果: 「${reward}」 を獲得しました！（もちものに追加）`;
  resultBox.classList.remove('hidden');
}
